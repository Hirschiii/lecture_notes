u := 1cm;

def draw_lampe(expr x, y, r) =
	path c;
	c = fullcircle scaled (1 * u) shifted (x * u, y * u);
	fill c withcolor white;
	draw c;
	draw ((x * u, y * u) + (dir (135 + r) * 0.5u))--((x * u, y * u) - (dir (135 + r) * 0.5u));
	draw ((x * u, y * u) + (dir (45 + r) * 0.5u))--((x * u, y * u) - (dir (45 + r) * 0.5u));
enddef;

def draw_va(expr x, y, r) =
	path c;
	c = fullcircle scaled (1 * u) shifted (x * u, y * u);
	fill c withcolor white;
	draw c;
	label("V/A", (x * u, y * u));
enddef;

def draw_source(expr x, y, r) =
	path c, d;
	c = fullcircle scaled (0.2 * u) shifted ((x + 0.5) * u, y * u);
	d = fullcircle scaled (0.2 * u) shifted ((x - 0.5) * u, y * u);
	fill c withcolor white;
	fill d withcolor white;
	draw c;
	draw d;
	label.top("$+$", ((x + 0.5) * u, (y + 0.2) * u));
	label.top("$-$", ((x - 0.5) * u, (y + 0.2) * u)); 
enddef;

def draw_led(expr x, y, r) =

	path d, c;
	c = fullcircle scaled (1 * u) shifted (x * u, y * u);

	d = ((x - 0.15) * u, (y + 0.3) * u)--((x - 0.15) * u, (y - 0.3) * u)--((x + 0.15) * u, y * u)--cycle;

	fill c withcolor white;
	fill d withcolor black;
	draw d;
	draw c;

	draw ((x + 0.15) * u, (y + 0.3) * u)--((x + 0.15) * u, (y - 0.3) * u) withpen pensquare scaled .5mm;
