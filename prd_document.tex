\environment env_document_multiple

\startluacode
	local path = environment.argument("path")

	local lfs = require("lfs")

	-- Table to hold filenames
	local files = {}

	-- Capture all filenames into the table
	for file in lfs.dir(path) do
		if file ~= "." and file ~= ".." then
			table.insert(files, file)
		end
	end

	-- Sort the filenames based on the numerical or date prefix
	table.sort(files, function(a, b)
			return a < b
	end)

	for _, file in ipairs(files) do
		if file:match("%.tex$") then
			context("\\input " .. path  .. file)
		end
	end

\stopluacode

\stopbodymatter

\component c_definitions
\component c_bibliography
\stopbackmatter
