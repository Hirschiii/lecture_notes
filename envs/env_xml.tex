startluacode
function getmeas(s)
if string.find(s, "[^0-9]") then
s = s:sub(1,-2)
s = s / 100
s = s.."\\textwidth"
tex.sprint(tex.ctxcatcodes, s)
else
s = s.."pt"
tex.sprint(s)
end
end
\stopluacode
